  AEROLINEAS {
    int aerolinea_id PK
    varchar nombre
    char codigo_IATA
    varchar pais_origen
    date fecha_fundacion
    varchar sitio_web
    timestamp myLogUP
  }
  AERONAVES {
    int aeronave_id PK
    int aerolinea_id FK
    varchar modelo
    int capacidad_pasajeros
    decimal capacidad_carga
    year fecha_fabricacion
    date ultimo_mantenimiento
    enum estado
    timestamp myLogUP
  }
  RUTAS {
    int ruta_id PK
    char aeropuerto_origen
    char aeropuerto_destino
    decimal distancia_km
    int tiempo_estimado_min
    timestamp myLogUP
  }
  VUELOS {
    int vuelo_id PK
    int aerolinea_id FK
    int ruta_id FK
    int aeronave_id FK
    timestamp fecha_salida
    timestamp fecha_llegada_estimada
    timestamp fecha_llegada_real
    varchar puerta_embarque
    int asientos_disponibles
    enum estado
    timestamp myLogUP
  }
  PASAJEROS {
    int pasajero_id PK
    varchar pasaporte
    varchar nombre
    varchar apellido
    date fecha_nacimiento
    varchar nacionalidad
    varchar telefono
    varchar email
    varchar programa_fidelidad
    int vuelos_acumulados
    timestamp myLogUP
  }
  TRIPULACION {
    int empleado_id PK
    int aerolinea_id FK
    varchar licencia
    varchar nombre
    varchar apellido
    enum puesto
    date fecha_contratacion
    decimal horas_vuelo
    timestamp myLogUP
  }
  RESERVACIONES {
    int reservacion_id PK
    int vuelo_id FK
    int pasajero_id FK
    enum clase
    varchar asiento
    timestamp fecha_reservacion
    enum estado
    decimal precio
    enum metodo_pago
    timestamp myLogUP
  }
  TRIPULACION_VUELO {
    int vuelo_id FK
    int empleado_id FK
    varchar rol
    timestamp myLogUP
  }
  PAGOS {
    int pago_id PK
    int reservacion_id FK
    decimal monto
    enum metodo_pago
    enum estado
    timestamp fecha_pago
    varchar referencia
    timestamp myLogUP
  }
  NOTIFICACIONES {
    int notificacion_id PK
    int pasajero_id FK
    int vuelo_id FK
    enum tipo
    varchar mensaje
    varchar email_destino
    tinyint enviada
    timestamp fecha_creacion
    timestamp myLogUP
  }
